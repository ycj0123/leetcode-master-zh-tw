# 787. K 站中轉內最便宜的航班 

有 n 個城市透過一些航班連接。給你一個陣列（array） flights ，其中 flights[i] = [fromi, toi, pricei] ，表示該航班從城市 fromi 出發，以價格 pricei 抵達 toi。

現在給定所有的城市和航班，以及出發城市 src 和目的地 dst，你的任務是找出一條最多經過 k 站中轉的路線，使得從 src 到 dst 的價格最便宜，並回傳該價格。如果不存在這樣的路線，則輸出 -1。

## 解題思路 



```CPP 
class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        vector<int> minDist(n , INT_MAX/2);
        minDist[src] = 0;
        vector<int> minDist_copy(n); // 用來記錄每一次遍歷的結果
        for (int i = 1; i <= k + 1; i++) {
            minDist_copy = minDist; // 取得上一次計算的結果
            for (auto &f : flights) {
                int from = f[0];
                int to = f[1];
                int price = f[2];
                minDist[to] = min(minDist_copy[from] + price, minDist[to]);
                // if (minDist[to] > minDist_copy[from] + price) minDist[to] = minDist_copy[from] + price;
            }

        }
        int result = minDist[dst] == INT_MAX/2 ? -1 : minDist[dst];
        return result;
    }
};
```

    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {

            int to = f[1];

        k++;
        queue<int> que;

class Solution {
struct Edge {
    int to;  // 連接的節點
    int val; // 邊的權重

    Edge(int t, int w): to(t), val(w) {}  // 建構函式
};

public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        vector<int> minDist(n , INT_MAX/2);
        vector<list<Edge>> grid(n + 1); // 相鄰串列
        for (auto &f : flights) {
            int from = f[0];
            int to = f[1];
            int price = f[2];
            grid[from].push_back(Edge(to, price));

        }
        minDist[src] = 0;
        vector<int> minDist_copy(n); // 用來記錄每一次遍歷的結果
        k++;
        queue<int> que;
        que.push(src);
        int que_size;
        while (k-- && !que.empty()) {
            // 注意這個陣列放的位置
            vector<bool> visited(n + 1, false); // 可加，可不加，加了效率高一些，防止佇列裡重複造訪，其數值已經算過了
            minDist_copy = minDist; // 取得上一次計算的結果
            que_size = que.size();
            while (que_size--) {
                int node = que.front(); que.pop();
                for (Edge edge : grid[node]) {
                    int from = node;
                    int to = edge.to;
                    int price = edge.val;
                    if (minDist[to] > minDist_copy[from] + price) {
                        minDist[to] = minDist_copy[from] + price;
                        if(visited[to]) continue; // 不用重複放入佇列，但需要重複計算，所以放在這個位置
                        visited[to] = true;
                        que.push(to);
                    }
                }

            }
        }
        int result = minDist[dst] == INT_MAX/2 ? -1 : minDist[dst];
        return result;
    }
};
