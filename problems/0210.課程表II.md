class Solution {
public:
    vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<int> inDegree(numCourses, 0);
        vector<int> result;
        unordered_map<int, vector<int>> umap;
        for (int i = 0; i < prerequisites.size(); i++) {

            // prerequisites[i][0] 是課程輸入度，prerequisites[i][1] 是課程外度數
            // 即：上課 prerequisites[i][0] 之前，必須先上課 prerequisites[i][1]
            // prerequisites[i][1] -> prerequisites[i][0]
            inDegree[prerequisites[i][0]]++;//當前課程輸入度值+1
            umap[prerequisites[i][1]].push_back(prerequisites[i][0]); // 添加 prerequisites[i][1] 指向的課程
        }
        queue<int> que;
        for (int i = 0; i < numCourses; i++) {
            if (inDegree[i] == 0) que.push(i); // 所有輸入度為 0，即為開頭課程加入佇列
        }
        int count = 0;
        while (que.size()) {
            int cur = que.front();  //當前選的課
            que.pop();
            count++; // 選課數+1 
            result.push_back(cur);
            vector<int> courses = umap[cur]; //取得這門課指向的課程，也就是這門課的後續課
            if (courses.size()) { // 有後續課
                for (int i = 0; i < courses.size(); i++) {
                    inDegree[courses[i]]--; // 它的後續課的輸入度-1
                    if (inDegree[courses[i]] == 0) que.push(courses[i]); // 如果輸入度為 0，加入佇列
                }
            }
        }
        if (count == numCourses) return result;
        else return vector<int>();
    }
};
