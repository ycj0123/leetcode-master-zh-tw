
# 44. 開發商購買土地 

> 本題為程式碼隨想錄後續擴充題目，還沒有影片講解，順便讓大家練習一下 ACM 輸入輸出模式（筆試面試必備）

[題目連結](https://kamacoder.com/problempage.php?pid=1044)

【題目描述】

在一個城市區域內，被劃分成了 n * m 個連續的區塊，每個區塊都擁有不同的權重（weight），代表著其土地價值。目前，有兩家開發公司，A 公司和 B 公司，希望購買這個城市區域的土地。

現在，需要將這個城市區域的所有區塊分配給 A 公司和 B 公司。

然而，由於城市規劃的限制，只允許將區域按橫向或縱向劃分成兩個子區域，而且每個子區域都必須包含一個或多個區塊。 

為了確保公平競爭，你需要找到一種分配方式，使得 A 公司和 B 公司各自的子區域內的土地總價值之差最小。

注意：區塊不可再分。

【輸入描述】

第一行輸入兩個正整數（positive integer），代表 n 和 m。

接下來的 n 列，每列輸出 m 個正整數。

輸出描述

請輸出一個整數（integer），代表兩個子區域內土地總價值之間的最小差距。

【輸入範例】

```
3 3
1 2 3
2 1 3
1 2 3
```

【輸出範例】

0

【提示資訊】

如果將區域按照如下方式劃分：

```
1 2 | 3
2 1 | 3
1 2 | 3
```

兩個子區域內土地總價值之間的最小差距可以達到 0。

【資料範圍】：

* 1 <= n, m <= 100；
* n 和 m 不同時為 1。

## 解題思路 

看到本題，大家如果想蠻力（brute force）求解，應該是 n^3 的時間複雜性（time complexity）， 

一個 for 列舉（enumeration）分割線，巢狀兩個 for 去累加區間（interval）裡的和。 

如果本題要求任何兩列（row）（或者兩行（column））之間的數值總和，大家在[0058.區間和](./0058.區間和.md) 的基礎上應該知道怎麼求。  

就是 prefix sum 的思路，先統計好前 n 列的和 q[n]，如果要求矩陣（matrix）a 列到 b 列之間的總和，那麼就 q[b] - q[a - 1] 就好。 

至於為什麼是 a - 1，大家去看 [0058.區間和](./0058.區間和.md) 的分析，使用 prefix sum 要注意區間左右邊的開閉情況。 

本題也可以使用 prefix sum 的思路來求解，先將列方向和行方向的和求出來，這樣可以方便知道劃分的兩個區間的和。 

程式碼如下：


```CPP 
#include <iostream>
#include <vector>
#include <climits>

using namespace std;
int main () {
    int n, m;
    cin >> n >> m;
    int sum = 0;
    vector<vector<int>> vec(n, vector<int>(m, 0)) ;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> vec[i][j];
            sum += vec[i][j];
        }
    }
    // 統計橫向
    vector<int> horizontal(n, 0);
    for (int i = 0; i < n; i++) {
        for (int j = 0 ; j < m; j++) {
            horizontal[i] += vec[i][j];
        }
    }
    // 統計縱向
    vector<int> vertical(m , 0);
    for (int j = 0; j < m; j++) {
        for (int i = 0 ; i < n; i++) {
            vertical[j] += vec[i][j];
        }
    }
    int result = INT_MAX;
    int horizontalCut = 0;
    for (int i = 0 ; i < n; i++) {
        horizontalCut += horizontal[i];
        result = min(result, abs(sum - horizontalCut - horizontalCut));
    }
    int verticalCut = 0;
    for (int j = 0; j < m; j++) {
        verticalCut += vertical[j];
        result = min(result, abs(sum - verticalCut - verticalCut));
    }
    cout << result << endl;
}

```

時間複雜性：O(n^2)

其實本題可以在蠻力求解的基礎上，最佳化一下，就不用 prefix sum 了，在橫向（列方向）遍歷（traversal）的時候，遇到列末尾就統計一下，在縱向（行方向）遍歷的時候，遇到行末尾就統計一下。 

時間複雜性也是 O(n^2)

程式碼如下：

```CPP 
#include <iostream>
#include <vector>
#include <climits>

using namespace std;
int main () {
    int n, m;
    cin >> n >> m;
    int sum = 0;
    vector<vector<int>> vec(n, vector<int>(m, 0)) ;
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> vec[i][j];
            sum += vec[i][j];
        }
    }

    int result = INT_MAX;
    int count = 0; // 統計遍歷過的列
    for (int i = 0; i < n; i++) {
        for (int j = 0 ; j < m; j++) {
            count += vec[i][j];
            // 遍歷到列末尾的時候開始統計
            if (j == m - 1) result = min (result, abs(sum - count - count));

        }
    }

    count = 0; // 統計遍歷過的行
    for (int j = 0; j < m; j++) {
        for (int i = 0 ; i < n; i++) {
            count += vec[i][j];
            // 遍歷到行末尾的時候開始統計
            if (i == n - 1) result = min (result, abs(sum - count - count));
        }
    }
    cout << result << endl;
}
 
```

```js
function func() {
    const readline = require('readline')
    const rl = readline.createInterface({
        input: process.stdin,
        output: process.stdout
    })

    rl.on('line', function (line) {
        inputLines.push(line.trim())
    })

    rl.on('close', function () {
        let [n, m] = inputLines[0].split(" ").map(Number)
        let c = new Array(n).fill(0)
        let r = new Array(m).fill(0)
        let arr = new Array(n)
        let sum = 0//陣列總和
        let min = Infinity//設定最小值的初始值為無限大
        //定義陣列
        for (let s = 0; s < n; s++) {
            arr[s] = inputLines[s + 1].split(" ").map(Number)
        }
        //每一列的和
        for (let i = 0; i < n; i++) {
            for (let j = 0; j < m; j++) {
                c[i] += arr[i][j]
                sum += arr[i][j]
            }
        }
        //每一行的和
        for (let i = 0; i < n; i++) {
            for (let j = 0; j < m; j++) {
                r[j] += arr[i][j]
            }
        }
        let sum1 = 0, sum2 = 0
        //橫向切割
        for (let i = 0; i < n; i++) {
            sum1 += c[i]
            min = min < Math.abs(sum - 2 * sum1) ? min : Math.abs(sum - 2 * sum1)
        }
        //縱向切割
        for (let j = 0; j < m; j++) {
            sum2 += r[j]
```

### C

prefix sum
```c
#include <stdlib.h>
#include <stdio.h>

int main()
{
    int n = 0, m = 0, ret_ver = 0, ret_hor = 0;

    // 讀取列和行的值
    scanf("%d%d", &n, &m);
    // 動態分配陣列 a(橫) 和 b(縱) 的空間
    int *a = (int *)malloc(sizeof(int) * n);
    int *b = (int *)malloc(sizeof(int) * m);

    // 初始化陣列 a 和 b
    for (int i = 0; i < n; i++)
    {
        a[i] = 0;
    }
    for (int i = 0; i < m; i++)
    {
        b[i] = 0;
    }

    // 讀取區塊權重並計算每列和每行的總權重
    for (int i = 0; i < n; i++)
    {
        for (int j = 0; j < m; j++)
        {
            int tmp;
            scanf("%d", &tmp);
            a[i] += tmp;
            b[j] += tmp;
        }
    }

    // 計算每行以及每列的 prefix sum
    for (int i = 1; i < n; i++)
    {
        a[i] += a[i - 1];
    }
    for (int i = 1; i < m; i++)
    {
        b[i] += b[i - 1];
    }

    // 初始化 ret_ver 和 ret_hor 為最大可能值
    ret_hor = a[n - 1];
    ret_ver = b[m - 1];

    // 計算按列劃分的最小差異
    int ret2 = 0;
    while (ret2 < n)
    {
        ret_hor = (ret_hor > abs(a[n - 1] - 2 * a[ret2])) ? abs(a[n - 1] - 2 * a[ret2]) : ret_hor;
        // 原理同行，但更進階
        ret2++;
    }
    // 計算按行劃分的最小差異
    int ret1 = 0;
    while (ret1 < m)
    {
        if (ret_ver > abs(b[m - 1] - 2 * b[ret1]))
        {
            ret_ver = abs(b[m - 1] - 2 * b[ret1]);
        }
        ret1++;
    }

    // 輸出最小差異
    printf("%d\n", (ret_ver <= ret_hor) ? ret_ver : ret_hor);

    // 釋放分配的記憶體
    free(a);
    free(b);
    return 0;
}
```

### Go

prefix sum

```go
package main

import (
    "fmt"
    "os"
    "bufio"
    "strings"
    "strconv"
    "math"
)
    
func main() {
    var n, m int
    
    reader := bufio.NewReader(os.Stdin)
    
    line, _ := reader.ReadString('\n')
    line = strings.TrimSpace(line)
    params := strings.Split(line, " ")
    
    n, _ = strconv.Atoi(params[0])
    m, _ = strconv.Atoi(params[1])//n 和 m 讀取完成
    
    land := make([][]int, n)//土地矩陣初始化
    
    for i := 0; i < n; i++ {
        line, _ := reader.ReadString('\n')
        line = strings.TrimSpace(line)
        values := strings.Split(line, " ")
        land[i] = make([]int, m)
        for j := 0; j < m; j++ {
            value, _ := strconv.Atoi(values[j])
            land[i][j] = value
        }
    }//所有讀取完成
    
    //初始化 prefix sum 矩陣
    preMatrix := make([][]int, n+1)
	for i := 0; i <= n; i++ {
		preMatrix[i] = make([]int, m+1)
	}
    
    for a := 1; a < n+1; a++ {
        for b := 1; b < m+1; b++ {
            preMatrix[a][b] = land[a-1][b-1] + preMatrix[a-1][b] + preMatrix[a][b-1] - preMatrix[a-1][b-1]
        }
    }
    
    totalSum := preMatrix[n][m]
    
    minDiff := math.MaxInt32//初始化極大數，用於比較
    
    //按列分割
    for i := 1; i < n; i++ {
        topSum := preMatrix[i][m]
        
        bottomSum := totalSum - topSum
        
        diff := int(math.Abs(float64(topSum - bottomSum)))
        if diff < minDiff {
            minDiff = diff
        }
    }
    
    //按行分割
    for j := 1; j < m; j++ {
        topSum := preMatrix[n][j]
        
        bottomSum := totalSum - topSum
        
        diff := int(math.Abs(float64(topSum - bottomSum)))
        if diff < minDiff {
            minDiff = diff
        }
    }    
    
    fmt.Println(minDiff) 
}
```
