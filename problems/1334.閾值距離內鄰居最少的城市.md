class Solution {
public:
    int findTheCity(int n, vector<vector<int>>& edges, int distanceThreshold) {
        vector<vector<int>> grid(n, vector<int>(n, 10005));  // 因為邊的最大距離是 10^4

        // 節點到自己的距離為 0
        for (int i = 0; i < n; i++) grid[i][i] = 0;
        // 建構相鄰矩陣
        for (const vector<int>& e : edges) {
            int from = e[0];
            int to = e[1];
            int val = e[2];
            grid[from][to] = val;
            grid[to][from] = val; // 注意這裡是雙向圖
        }

        // 開始 floyd
        // 思考 為什麼 p 要放在最外面一層
        for (int p = 0; p < n; p++) {
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    grid[i][j] = min(grid[i][j], grid[i][p] + grid[p][j]);
                }
            }
        }

        int result = 0;
        int count = n + 10; // 記錄所有城市在範圍內連接的最小城市數量
        for (int i = 0; i < n; i++) {
            int curCount = 0; // 統計一個城市在範圍內可以連接幾個城市
            for (int j = 0; j < n; j++) {
                if (i != j && grid[i][j] <= distanceThreshold) curCount++;
                // cout << "i:" << i << ", j:" << j << ", val: " << grid[i][j] << endl;
            }
            if (curCount <= count) { // 注意這裡是 <=
                count = curCount;
                result = i;
            }
        }
        return result;
    }
};
